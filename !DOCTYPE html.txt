    const drawBtn = document.getElementById("drawBtn");

    // 渲染网格方块
    function renderGrid() {
      gridDiv.innerHTML = "";
      issues.forEach((issue, i) => {
        let block = document.createElement("div");
        block.className = "block";
        block.style.background = colors[i % colors.length];
        block.innerHTML = `<span class="num">#${i+1}</span><div class="short">${issue.short}</div>`;
        
        if (progress.today === i) block.classList.add("today");
        if (progress.done.includes(i)) block.classList.add("done");
        
        block.onclick = () => showEssay(i);
        gridDiv.appendChild(block);
      });
      updateProgressBar();
    }
    
    // 更新进度条
    function updateProgressBar() {
      const total = issues.length;
      const done = progress.done.length;
      const pct = Math.round(done/total*100);
      progressBar.style.width = `${pct}%`;
      progressBar.textContent = `${done} / ${total} (${pct}%)`;
    }
    
    // 显示范文弹窗
    function showEssay(idx) {
      let issue = issues[idx];
      document.getElementById("outlineTitle").textContent = `题目 #${idx+1}: ${issue.short}`;
      document.getElementById("outlineContent").innerHTML =
        `<div style="margin-bottom:15px; padding:10px; background:#f9f9f9; border-radius:5px;"><b>题目：</b> ${issue.text}</div>` +
        `<div>${issue.essay}</div>`;
      
      document.getElementById("outlineBox").style.display = "block";
      overlay.style.display = "block";
      document.getElementById("doneBtn").style.display = progress.done.includes(idx) ? "none" : "inline-block";
      document.getElementById("mindmapBtn").style.display = "inline-block";
      document.getElementById("mindmapBtn").textContent = "显示结构性提纲";
      document.getElementById("mindmapBtn").onclick = () => showOutline(idx);
      
      // 更新今日题目
      if (progress.today !== idx) {
        progress.today = idx;
        localStorage.setItem("gre_grid_draw_mindmap_progress", JSON.stringify(progress));
        renderGrid();
      }
    }
    
    // 显示结构性提纲
    function showOutline(idx) {
      let issue = issues[idx];
      document.getElementById("outlineTitle").textContent = `题目 #${idx+1} 的结构性提纲`;
      document.getElementById("outlineContent").innerHTML =
        `<div style="margin-bottom:15px; padding:10px; background:#f9f9f9; border-radius:5px;"><b>题目：</b> ${issue.text}</div>` +

      document.getElementById("mindmapBtn").textContent = "显示思维导图";
      document.getElementById("mindmapBtn").onclick = () => showMindmap(idx);
    }
    
    // 显示思维导图
    function showMindmap(idx) {
      let issue = issues[idx];
      let html = `<div style="margin-bottom:15px; padding:10px; background:#f9f9f9; border-radius:5px;"><b>题目：</b> ${issue.text}</div>`;
      html += `<div class="mindmap">`;
      issue.mindmap.forEach(node => {
        html += `<div class="node">${node.label}`;
        node.subs.forEach(sub => {
          html += `<div class="subnode">${sub}</div>`;
        });
        html += `</div>`;
      });
      html += `</div>`;
      document.getElementById("outlineContent").innerHTML = html;
      document.getElementById("mindmapBtn").textContent = "显示结构性提纲";
      document.getElementById("mindmapBtn").onclick = () => showOutline(idx);
    }
    
    // 隐藏弹窗
    function hideOutline() {
      document.getElementById("outlineBox").style.display = "none";
      overlay.style.display = "none";
    }
    
    // 打卡完成
    document.getElementById("doneBtn").onclick = function() {
      if (!progress.done.includes(progress.today)) {
        progress.done.push(progress.today);
        localStorage.setItem("gre_grid_draw_mindmap_progress", JSON.stringify(progress));
        renderGrid();
        this.style.display = "none";
        hideOutline();
      }
    };
    
    // 抽题动画与分配
    drawBtn.onclick = function() {
      let available = issues.map((t, i) => i).filter(i => !progress.done.includes(i));
      if (available.length === 0) {
        alert("恭喜！所有题目已完成！");
        return;
      }
      
      drawBtn.disabled = true;
      rollingBox.style.display = "block";
      rollingBox.textContent = "抽选中 ...";
      
      let rollingOrder = [];
      for (let i = 0; i < 16; i++) {
        rollingOrder.push(available[Math.floor(Math.random() * available.length)]);
      }
      
      let finalIdx = rollingOrder[rollingOrder.length - 1];
      let step = 0;
      
      let rollingAnim = setInterval(() => {
        rollingBox.textContent = `抽选中: 题目 #${rollingOrder[step] + 1} ...`;
        step++;
        
        if (step >= rollingOrder.length) {
          clearInterval(rollingAnim);
          setTimeout(() => {
            rollingBox.style.display = "none";
            showEssay(finalIdx);
            drawBtn.disabled = false;
          }, 500);
        }
      }, 120);
    };

    // 添加键盘事件支持

      if (e.key === 'Escape') {
        hideOutline();
      }
    });
